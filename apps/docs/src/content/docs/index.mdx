## Write the plugin in C++ or Rust

A plugin is a class with a constructor and a `process()` method. You write the DSP. The SDK's header handles parameters, state, presets and the C interface the host calls. Parameter ranges and defaults come from `plugin.json`.

</div>

<Tabs syncKey="lang">
  <TabItem label="C++" icon="seti:cpp">

```cpp title="src/plugin.cpp"
#include "audiospan/effect.hpp"
#include "audiospan_params.h"  // generated from plugin.json

class GainEffect {
 public:
  explicit GainEffect(const audiospan::HostConfig& config)
      : params_(plugin::kParameters, config.sampleRate) {}

  audiospan::Parameters<plugin::kParamCount>& parameters() noexcept {
    return params_;
  }

  void process(const audiospan::ProcessContext& ctx) noexcept {
    for (uint32_t ch = 0; ch < ctx.outputChannels; ++ch) {
      // Per-sample ramp towards the smoothed target: click-free.
      auto gain = params_.ramp(plugin::kGain, ctx.frameCount);
      const float* in = ctx.input(ch);
      float* out = ctx.outputs[ch];
      for (uint32_t i = 0; i < ctx.frameCount; ++i)
        out[i] = in ? in[i] * static_cast<float>(gain.next()) : 0.0f;
    }
  }

 private:
  audiospan::Parameters<plugin::kParamCount> params_;
};

AUDIOSPAN_DEFINE_EFFECT(GainEffect, 16)
```

```rust title="src/lib.rs"
pub mod abi;
#[path = "../generated/audiospan_params.rs"] // generated from plugin.json
mod params;

use abi::{Effect, HostConfig, Parameters, ProcessContext};

pub struct GainEffect {
    params: Parameters,
}

impl Effect for GainEffect {
    fn create(config: &HostConfig) -> Option<Self> {
        let sample_rate = config.sample_rate as f64;
        let params = Parameters::new(&params::PARAMETERS, sample_rate);
        Some(Self { params })
    }

    fn parameters(&self) -> Option<&Parameters> {
        Some(&self.params)
    }
    fn parameters_mut(&mut self) -> Option<&mut Parameters> {
        Some(&mut self.params)
    }

    fn process(&mut self, ctx: &mut ProcessContext) {
        for ch in 0..ctx.output_channels() {
            // Per-sample ramp towards the smoothed target: click-free.
            let mut gain = self.params.ramp(params::GAIN, ctx.frame_count());
            if let Some((input, output)) = ctx.io(ch) {
                for (i, out) in output.iter_mut().enumerate() {
                    *out = input.map_or(0.0, |x| x[i]) * gain.next() as f32;
                }
            }
        }
    }
}

audiospan_define_effect!(GainEffect, 16);
```

## Use it from React

`audiospan create --app react` sets up a plugin and a Vite app together. `audiospan init` adds a plugin to a React project you already have. Either way you get a component like this one, and the plugin's parameter, output and preset ids are type-checked.

- `useAudiospanHost` creates the AudioContext and the host when the user clicks Start.
- `usePlugin` loads the plugin and keeps its values and meters in React state.
- `useAudioSource` connects a microphone, an `<audio>` element, a stream or any audio node.

</div>
<div class="as-split__code">

```tsx title="src/audiospan/Lowpass.tsx"
import {
  Meters, ParameterControls, useAudioSource, useAudiospanHost, usePlugin,
} from "@audiospan/react";
import type { LowpassPlugin } from "../plugins/lowpass.js"; // generated

const PLUGIN = "/plugins/lowpass/plugin.json";

export function Lowpass() {
  const audio = useAudiospanHost(); // owns the AudioContext and the host
  const plugin = usePlugin<LowpassPlugin>(audio.host, PLUGIN); // load + chain
  useAudioSource(audio, "microphone"); // or <audio>, a stream, a node
  return (
    <>
      <button onClick={() => void audio.start()}>Start audio</button>
      <ParameterControls plugin={plugin} />
      <Meters plugin={plugin} />
    </>
  );
}
```
